```python
def validate_batch(loss, batch, model):
    # NaN / Inf loss
    if torch.isnan(loss) or torch.isinf(loss):
        raise ValueError("[BATCH ERROR] NaN/Inf loss detected")

    # Loss explosion
    if loss.item() > 1e3:
        return "loss_explosion"

    # Empty batch
    if batch is None or len(batch["input_ids"]) == 0:
        raise ValueError("[BATCH ERROR] Empty batch detected")

    return "ok"
```

### Step 2 — Gradient anomaly detection

Run **after backward**:

```python
def check_gradients(model):
    total_norm = 0.0

    for p in model.parameters():
        if p.grad is not None:
            if torch.isnan(p.grad).any():
                raise ValueError("[GRAD ERROR] NaN gradients detected")

            param_norm = p.grad.data.norm(2)
            total_norm += param_norm.item() ** 2

    total_norm = total_norm ** 0.5

    if total_norm > 1e3:
        return "grad_explosion"

    return total_norm
```
