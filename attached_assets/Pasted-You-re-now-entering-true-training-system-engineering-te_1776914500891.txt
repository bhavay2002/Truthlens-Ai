```python
import torch

class AnomalyClassifier:
    def __init__(
        self,
        spike_ratio=2.5,
        explode_th=1000,
        vanish_th=1e-7,
        var_th=5.0
    ):
        self.spike_ratio = spike_ratio
        self.explode_th = explode_th
        self.vanish_th = vanish_th
        self.var_th = var_th

    def classify(
        self,
        loss,
        ema_loss,
        grad_stats,
        loss_var=None,
        logits=None,
        labels=None
    ):
        # --- 1. Hard failures ---
        if not torch.isfinite(torch.tensor(loss)):
            return "nan_loss"

        if logits is not None and not torch.isfinite(logits).all():
            return "nan_logits"

        # --- 2. Gradient pathologies ---
        grad_norm = grad_stats.get("total_norm", 0)

        if grad_norm > self.explode_th:
            return "exploding_gradients"

        if grad_norm < self.vanish_th:
            return "vanishing_gradients"

        # --- 3. Data / label issues ---
        if labels is not None:
            if (labels < 0).any():
                return "negative_labels"

        # --- 4. Instability ---
        ratio = loss / (ema_loss + 1e-8)

        if loss_var is not None and loss_var > self.var_th:
            return "high_variance"

        if ratio > self.spike_ratio:
            return "loss_spike"

        # --- 5. Logit collapse (silent failure) ---
        if logits is not None:
            if logits.std().item() < 1e-4:
                return "logit_collapse"

        return "unknown"
```

## ✅ Proper GradNorm (Paper-aligned)

```python
class GradNorm:
    def __init__(self, tasks, alpha=0.5, eps=1e-8):
        self.alpha = alpha
        self.eps = eps
        self.tasks = tasks

        self.initial_losses = None
        self.weights = {t: 1.0 for t in tasks}

    def compute(
        self,
        task_losses,
        task_grad_norms
    ):
        if self.initial_losses is None:
            self.initial_losses = {t: float(task_losses[t]) for t in self.tasks}

        # --- Normalize losses ---
        loss_ratios = {
            t: task_losses[t] / (self.initial_losses[t] + self.eps)
            for t in self.tasks
        }

        avg_ratio = sum(loss_ratios.values()) / len(loss_ratios)

        # --- Target gradient norms ---
        targets = {
            t: avg_ratio * (loss_ratios[t] ** self.alpha)
            for t in self.tasks
        }

        # --- Compute weights ---
        new_weights = {}
        for t in self.tasks:
            g = task_grad_norms[t] + self.eps
            new_weights[t] = targets[t] / g

        # --- Normalize weights (IMPORTANT) ---
        total = sum(new_weights.values())
        num_tasks = len(self.tasks)

        for t in new_weights:
            new_weights[t] = new_weights[t] * num_tasks / (total + self.eps)

        self.weights = new_weights
        return new_weights
```

```python
def compute_task_grad_norms(model, losses, shared_params):
```

```python
class GradHookManager:
    def __init__(self):
        self.buffer = {}

    def attach(self, model, filter_fn=None):
        for name, p in model.named_parameters():
            if not p.requires_grad:
                continue

            if filter_fn and not filter_fn(name):
                continue

            p.register_hook(self._hook(name))

    def _hook(self, name):
        def fn(grad):
            if grad is None:
                return

            norm = grad.norm().item()

            if name not in self.buffer:
                self.buffer[name] = []

            self.buffer[name].append(norm)

        return fn

    def aggregate(self):
        agg = {}

        for name, values in self.buffer.items():
            agg[name] = {
                "mean": sum(values) / len(values),
                "max": max(values),
                "min": min(values)
            }

        return agg

    def reset(self):
        self.buffer = {}
```

## ✅ Robust Version (with EMA + ratio)

```python
class TaskDominanceDetector:
    def __init__(self, alpha=0.1, dominance_ratio=5.0):
        self.alpha = alpha
        self.ratio = dominance_ratio
        self.ema = {}

    def update(self, task_grads):
        # --- EMA smoothing ---
        for t, g in task_grads.items():
            if t not in self.ema:
                self.ema[t] = g
            else:
                self.ema[t] = self.alpha * g + (1 - self.alpha) * self.ema[t]

        max_task = max(self.ema, key=self.ema.get)
        min_task = min(self.ema, key=self.ema.get)

        if self.ema[min_task] == 0:
            return None

        if self.ema[max_task] / self.ema[min_task] > self.ratio:
            return {
                "dominant": max_task,
                "suppressed": min_task,
                "ratio": self.ema[max_task] / self.ema[min_task]
            }

        return None
```
