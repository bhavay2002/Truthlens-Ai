# ✅ 1. Robust Mask + Clean Labels (vector-safe)

### ❗ Key rule:

* **Never use raw labels directly**
* Always derive **mask first**, then clean

```python
def prepare_targets(labels_dict):
    """
    labels_dict: {head: tensor [B]}
    returns:
        clean_labels: {head: tensor}
        masks: {head: bool tensor}
    """
    clean_labels = {}
    masks = {}

    for head, y in labels_dict.items():
        mask = ~torch.isnan(y)
        y_clean = torch.where(mask, y, torch.zeros_like(y))  # safe replacement

        clean_labels[head] = y_clean.long()  # ensure correct dtype
        masks[head] = mask

    return clean_labels, masks
```

# ✅ 2. Masked Multi-Head Loss (fully correct)

### 🔥 Handles:

* zero-label batches
* per-head masking
* class imbalance hooks
* safe normalization

```python
def compute_multitask_loss(outputs, labels, criteria, head_weights=None):
    """
    outputs: {head: logits [B, C]}
    labels: {head: tensor [B] with NaN}
    criteria: {head: loss_fn}
    head_weights: optional {head: float}
    """

    clean_labels, masks = prepare_targets(labels)

    total_loss = 0.0
    active_heads = 0
    per_head_loss = {}

    for head in outputs.keys():
        logits = outputs[head]
        target = clean_labels[head]
        mask = masks[head]

        valid_count = mask.sum()

        if valid_count == 0:
            per_head_loss[head] = None
            continue  # 🔴 critical skip

        # Apply mask
        logits_masked = logits[mask]
        target_masked = target[mask]

        loss_fn = criteria[head]
        loss_head = loss_fn(logits_masked, target_masked)

        # Optional weighting
        weight = head_weights.get(head, 1.0) if head_weights else 1.0
        loss_head = loss_head * weight

        total_loss += loss_head
        active_heads += 1
        per_head_loss[head] = loss_head.detach()

    # 🔥 Safe normalization
    if active_heads == 0:
        raise RuntimeError("No active heads in batch — all labels are NaN")

    total_loss = total_loss / active_heads

    return total_loss, per_head_loss, active_heads
```

from sklearn.metrics import f1_score

def compute_metrics(outputs, labels):
    metrics = {}

    for head in outputs:
        logits = outputs[head]
        y = labels[head]

        mask = ~torch.isnan(y)

        if mask.sum() == 0:
            metrics[head] = None
            continue

        preds = torch.argmax(logits[mask], dim=1).cpu().numpy()
        targets = y[mask].cpu().numpy()

        f1 = f1_score(targets, preds, average="macro")
        metrics[head] = f1

    return metrics
```

class HeadStats:

def training_step(batch, model, criteria, optimizer, head_weights=None):
    model.train()

    inputs = batch["inputs"]
    labels = batch["labels"]  # dict of heads

    outputs = model(inputs)

    loss, per_head_loss, active_heads = compute_multitask_loss(
        outputs, labels, criteria, head_weights
    )

    optimizer.zero_grad()
    loss.backward()
    optimizer.step()

    metrics = compute_metrics(outputs, labels)

    return {
        "loss": loss.item(),
        "per_head_loss": per_head_loss,
        "active_heads": active_heads,
        "metrics": metrics
    }
```
