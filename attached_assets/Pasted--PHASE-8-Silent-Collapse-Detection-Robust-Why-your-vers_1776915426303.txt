import numpy as np

class SilentCollapseDetector:
    def __init__(
        self,
        loss_alpha=0.1,
        f1_alpha=0.1,
        loss_drop_ratio=0.5,
        f1_floor=0.1,
        patience=5
    ):
        self.loss_ema = None
        self.f1_ema = None

        self.loss_alpha = loss_alpha
        self.f1_alpha = f1_alpha

        self.loss_drop_ratio = loss_drop_ratio
        self.f1_floor = f1_floor
        self.patience = patience

        self.counter = 0

    def update(self, loss, f1):
        # --- EMA smoothing ---
        if self.loss_ema is None:
            self.loss_ema = loss
            self.f1_ema = f1
            return False

        self.loss_ema = self.loss_alpha * loss + (1 - self.loss_alpha) * self.loss_ema
        self.f1_ema = self.f1_alpha * f1 + (1 - self.f1_alpha) * self.f1_ema

        # --- Collapse logic ---
        loss_condition = loss < self.loss_drop_ratio * self.loss_ema
        f1_condition = self.f1_ema < self.f1_floor

        if loss_condition and f1_condition:
            self.counter += 1
        else:
            self.counter = 0

        return self.counter >= self.patience

def classify_collapse_type(logits, labels):
    if logits is None:
        return "unknown"

    probs = torch.softmax(logits, dim=-1)

    # All predictions same class → mode collapse
    preds = probs.argmax(dim=-1)
    if preds.unique().numel() == 1:
        return "mode_collapse"

    # Extremely low confidence
    if probs.max().item() < 0.4:
        return "confidence_collapse"

    return "unknown"

class BatchAnalyzer:
    def __init__(self):
        self.priority = [
            "nan_loss",
            "label_error",
            "grad_explosion",
            "silent_collapse",
            "loss_spike",
            "high_variance"
        ]

    def analyze(self, stats: dict):
        flags = []

        if stats.get("nan_loss"):
            flags.append("nan_loss")

        if stats.get("label_invalid"):
            flags.append("label_error")

        if stats.get("grad_norm", 0) > 1000:
            flags.append("grad_explosion")

        if stats.get("silent_collapse"):
            flags.append("silent_collapse")

        if stats.get("loss_spike"):
            flags.append("loss_spike")

        if stats.get("loss_var", 0) > 5:
            flags.append("high_variance")

        if not flags:
            return "normal"

        # Return highest priority issue
        for p in self.priority:
            if p in flags:
                return p

        return flags[0]
