```python
import math

class TaskDominanceDetector:
    def __init__(self, ratio_thresh=5.0, alpha=0.1, eps=1e-8):
        self.ratio_thresh = ratio_thresh
        self.alpha = alpha
        self.eps = eps
        self.ema = {}

    def update(self, task_grads: dict, task_losses: dict = None):
        # --- EMA smoothing ---
        for t, g in task_grads.items():
            g = float(g)

            if t not in self.ema:
                self.ema[t] = g
            else:
                self.ema[t] = self.alpha * g + (1 - self.alpha) * self.ema[t]

        # --- Compute dominance ---
        max_t = max(self.ema, key=self.ema.get)
        min_t = min(self.ema, key=self.ema.get)

        min_val = self.ema[min_t]

        if min_val < self.eps:
            return {
                "dominant": max_t,
                "suppressed": min_t,
                "ratio": math.inf,
                "type": "grad_zero_collapse"
            }

        ratio = self.ema[max_t] / (min_val + self.eps)

        if ratio > self.ratio_thresh:
            return {
                "dominant": max_t,
                "suppressed": min_t,
                "ratio": ratio,
                "type": "grad_dominance"
            }

        return None
```

```python
class SilentCollapseDetector:
    def __init__(
        self,
        loss_ratio=0.5,
        metric_floor=0.1,
        alpha=0.1,
        patience=5
    ):
        self.loss_ema = None
        self.metric_ema = None

        self.loss_ratio = loss_ratio
        self.metric_floor = metric_floor
        self.alpha = alpha
        self.patience = patience

        self.counter = 0

    def update(self, loss, metric):
        if self.loss_ema is None:
            self.loss_ema = loss
            self.metric_ema = metric
            return False

        self.loss_ema = self.alpha * loss + (1 - self.alpha) * self.loss_ema
        self.metric_ema = self.alpha * metric + (1 - self.alpha) * self.metric_ema

        loss_condition = loss < self.loss_ratio * self.loss_ema
        metric_condition = self.metric_ema < self.metric_floor

        if loss_condition and metric_condition:
            self.counter += 1
        else:
            self.counter = 0

        return self.counter >= self.patience
```

```python
class GradientConflictDetector:
    def __init__(self, threshold=0.0):
        self.threshold = threshold

    def compute(self, task_grad_vecs: dict):
        conflicts = {}
        tasks = list(task_grad_vecs.keys())

        for i in range(len(tasks)):
            for j in range(i + 1, len(tasks)):
                t1, t2 = tasks[i], tasks[j]

                g1 = task_grad_vecs[t1]
                g2 = task_grad_vecs[t2]

                if g1.numel() == 0 or g2.numel() == 0:
                    continue

                cos_sim = F.cosine_similarity(g1, g2, dim=0).item()

                if cos_sim < self.threshold:
                    conflicts[(t1, t2)] = cos_sim

        return conflicts
```

```python
import torch.nn as nn

class TaskBalancer(nn.Module):
    def __init__(self, tasks):
        super().__init__()
        self.log_vars = nn.ParameterDict({
            t: nn.Parameter(torch.zeros(1)) for t in tasks
        })

    def uncertainty_weight(self, loss, task):
        log_var = self.log_vars[task]
        precision = torch.exp(-log_var)

        return precision * loss + log_var

    def forward(self, task_losses):
        total = 0
        for t, loss in task_losses.items():
            total += self.uncertainty_weight(loss, t)
        return total
```

```python
class AutoDebugEngine:
    def __init__(self, detectors, classifier, memory):
        self.detectors = detectors
        self.classifier = classifier
        self.memory = memory

    def step(self, signals):
        results = {}

        # --- Run detectors ---
        for name, detector in self.detectors.items():
            if hasattr(detector, "update"):
                results[name] = detector.update(**signals)
            else:
                results[name] = detector(**signals)

        # --- Classify ---
        failure_type = self.classifier.classify(results)

        # --- Store memory ---
        if failure_type != "normal":
            self.memory.store({
                "type": failure_type,
                "signals": signals,
                "detector_outputs": results
            })

        return failure_type, results
```

```python
class FailureMemory:
    def __init__(self, max_size=1000):
        self.buffer = []
        self.max_size = max_size

    def store(self, record):
        self.buffer.append(record)

        if len(self.buffer) > self.max_size:
            self.buffer.pop(0)

    def summary(self):
        from collections import Counter
        return Counter([r["type"] for r in self.buffer])
```
