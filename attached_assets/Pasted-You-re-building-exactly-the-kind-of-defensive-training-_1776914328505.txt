## ✅ Fixed + Extended Version

```python
import torch

class LossTracker:
    def __init__(self, tasks, alpha=0.1, eps=1e-8):
        self.alpha = alpha
        self.eps = eps
        self.ema = {t: None for t in tasks}
        self.steps = {t: 0 for t in tasks}

    def update(self, losses: dict):
        smoothed = {}

        for t, val in losses.items():
            val = float(val)

            if not torch.isfinite(torch.tensor(val)):
                raise ValueError(f"Non-finite loss detected in task {t}: {val}")

            self.steps[t] += 1

            if self.ema[t] is None:
                self.ema[t] = val
            else:
                self.ema[t] = self.alpha * val + (1 - self.alpha) * self.ema[t]

            # Bias correction (important early)
            bias_correction = 1 - (1 - self.alpha) ** self.steps[t]
            corrected = self.ema[t] / (bias_correction + self.eps)

            smoothed[t] = corrected

        return smoothed
```

## 🔬 Add: Loss Variance (VERY IMPORTANT)

Detect instability even when mean looks fine.

```python
class LossStats:
    def __init__(self, tasks, window=50):
        self.window = window
        self.history = {t: [] for t in tasks}

    def update(self, losses):
        stats = {}

        for t, val in losses.items():
            hist = self.history[t]
            hist.append(float(val))

            if len(hist) > self.window:
                hist.pop(0)

            if len(hist) > 1:
                var = float(torch.tensor(hist).var())
            else:
                var = 0.0

            stats[t] = {
                "mean": sum(hist) / len(hist),
                "var": var
            }

        return stats
```

```python
class GradTracker:
    def __init__(self, window=50):
        self.window = window
        self.history = []

    def update(self, model):
        total_norm = 0.0
        norms = []
        vars_ = []

        for name, p in model.named_parameters():
            if p.grad is None:
                continue

            g = p.grad.detach()

            norm = g.norm().item()
            var = g.var().item()

            norms.append(norm)
            vars_.append(var)

            total_norm += norm ** 2

        total_norm = total_norm ** 0.5

        record = {
            "total_norm": total_norm,
            "mean_norm": sum(norms) / (len(norms) + 1e-8),
            "mean_var": sum(vars_) / (len(vars_) + 1e-8)
        }

        self.history.append(record)
        if len(self.history) > self.window:
            self.history.pop(0)

        return record
```

## ✅ Fixed Version (Relative + Z-score hybrid)

```python
class SpikeDetector:
    def __init__(self, threshold=2.5, eps=1e-8):
        self.threshold = threshold
        self.eps = eps

    def detect(self, loss, ema_loss, var=None):
        ratio = loss / (ema_loss + self.eps)

        if var is not None and var > 0:
            z = (loss - ema_loss) / (var ** 0.5 + self.eps)
        else:
            z = 0

        return ratio > self.threshold or z > self.threshold
```

```python
import os
import time

def dump_batch(debug_path, payload):
    os.makedirs(debug_path, exist_ok=True)

    safe_payload = {}

    for k, v in payload.items():
        if torch.is_tensor(v):
            safe_payload[k] = v.detach().cpu()
        else:
            safe_payload[k] = v

    filename = f"spike_{int(time.time() * 1000)}.pt"
    path = os.path.join(debug_path, filename)

    torch.save(safe_payload, path)

    return path
```
