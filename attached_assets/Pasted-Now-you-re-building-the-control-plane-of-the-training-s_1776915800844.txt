```python id="fc_pro"
class FailureClassifier:
    def __init__(self):
        # Priority order (top = most critical)
        self.priority = [
            "numerical_instability",
            "gradient_explosion",
            "representation_failure",
            "task_imbalance",
            "gradient_conflict",
            "persistent_instability"
        ]

    def classify(self, signals: dict):
        flags = []

        if signals.get("nan_loss") or signals.get("nan_logits"):
            flags.append("numerical_instability")

        if signals.get("grad_norm", 0) > 1000:
            flags.append("gradient_explosion")

        if signals.get("silent_collapse"):
            flags.append("representation_failure")

        if signals.get("dominance"):
            flags.append("task_imbalance")

        if signals.get("conflicts"):
            flags.append("gradient_conflict")

        if signals.get("spike_cluster"):
            flags.append("persistent_instability")

        if not flags:
            return "normal", []

        # Return highest priority + all contributing causes
        for p in self.priority:
            if p in flags:
                return p, flags

        return flags[0], flags
```

## ✅ Improved (EMA + density-aware)

```python id="spike_cluster_pro"
from collections import deque

class SpikeCluster:
    def __init__(self, window=50, spike_ratio=0.2, min_events=5):
        self.window = deque(maxlen=window)
        self.spike_ratio = spike_ratio
        self.min_events = min_events

    def update(self, is_spike: bool):
        self.window.append(1 if is_spike else 0)

        if len(self.window) < self.min_events:
            return False

        ratio = sum(self.window) / len(self.window)

        return ratio > self.spike_ratio

    def density(self):
        if not self.window:
            return 0.0
        return sum(self.window) / len(self.window)
```

```python id="memory_pro"
from collections import defaultdict, Counter
import time

class FailureMemory:
    def __init__(self, max_per_type=500):
        self.history = defaultdict(list)
        self.max_per_type = max_per_type

    def store(self, failure_type, signals):
        record = {
            "timestamp": time.time(),
            "signals": signals
        }

        self.history[failure_type].append(record)

        # Cap memory
        if len(self.history[failure_type]) > self.max_per_type:
            self.history[failure_type].pop(0)

    def get_stats(self):
        return {k: len(v) for k, v in self.history.items()}

    def recent(self, failure_type, n=5):
        return self.history[failure_type][-n:]

    def distribution(self):
        return Counter({
            k: len(v) for k, v in self.history.items()
        })
```

## ✅ Production Health Score

```python id="health_pro"
class HealthScore:
    def __init__(self):
        self.weights = {
            "spike": 0.15,
            "spike_cluster": 0.25,
            "dominance": 0.15,
            "conflicts": 0.15,
            "silent_collapse": 0.30
        }

    def compute(self, signals: dict):
        score = 1.0

        for key, weight in self.weights.items():
            if signals.get(key):
                score -= weight

        return max(score, 0.0)

    def interpret(self, score):
        if score >= 0.8:
            return "healthy"
        elif score >= 0.5:
            return "unstable"
        else:
            return "failing"
```

## 🔥 Add: Smoothed Health (VERY useful)

```python id="health_ema"
class SmoothedHealth:
    def __init__(self, alpha=0.1):
        self.alpha = alpha
        self.ema = None

    def update(self, score):
        if self.ema is None:
            self.ema = score
        else:
            self.ema = self.alpha * score + (1 - self.alpha) * self.ema
        return self.ema
```
