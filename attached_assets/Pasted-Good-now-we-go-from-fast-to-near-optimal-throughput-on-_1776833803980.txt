## ✅ B. Bucketed sampler (group similar lengths)

```python
from torch.utils.data import Sampler
import math, random

class BucketSampler(Sampler):
    def __init__(self, lengths, batch_size, shuffle=True, bucket_size=100):
        self.batch_size = batch_size
        self.shuffle = shuffle

        indices = list(range(len(lengths)))
        indices.sort(key=lambda i: lengths[i])

        # chunk into buckets
        self.buckets = [
            indices[i:i+bucket_size]
            for i in range(0, len(indices), bucket_size)
        ]

    def __iter__(self):
        if self.shuffle:
            for b in self.buckets:
                random.shuffle(b)
            random.shuffle(self.buckets)

        batch = []
        for bucket in self.buckets:
            for idx in bucket:
                batch.append(idx)
                if len(batch) == self.batch_size:
                    yield batch
                    batch = []

        if batch:
            yield batch

    def __len__(self):
        return math.ceil(sum(len(b) for b in self.buckets) / self.batch_size)
```
